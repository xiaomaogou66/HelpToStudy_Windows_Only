### 📚 学习块

```dataviewjs
const now = dv.date("today");

// 连续打卡：当天所有打卡项都勾上才算「这天打卡成功」（条目形如 "09-14 · 内容"）
// 计数规则（2026-09-14 定）：今天勾完 → 今天计入；今天还没勾完 → 数字不动
// （停在昨天为止的连续数），不会因为"今天还没勾"而归零。
const byDay = {};
for (const p of dv.pages('"05-日程安排/06-日志"')) {
  for (const t of p.file.tasks) {
    const m = String(t.text).match(/^(\d{2}-\d{2}) · /);
    if (m) (byDay[m[1]] = byDay[m[1]] || []).push(t.completed);
  }
}
const todayArr = byDay[now.toFormat("MM-dd")] || [];
const todayDone = todayArr.length > 0 && !todayArr.includes(false);

let n = 0, d = todayDone ? now : now.minus({ days: 1 });
while (true) {
  const arr = byDay[d.toFormat("MM-dd")];
  if (!arr || arr.length === 0 || arr.includes(false)) break;   // 没勾完 / 那天没有条目 → 断
  n += 1;
  d = d.minus({ days: 1 });
}
const doneToday = todayArr.filter(Boolean).length;

// 四道倒计时（读 05-日程安排/00-配置/规划配置.json 的「里程碑」——唯一真源）
let milestones = [];
try {
  const _cfg = JSON.parse(await dv.app.vault.adapter.read("05-日程安排/00-配置/规划配置.json"));
  milestones = (_cfg["里程碑"] || []).map(m => [m["名称"], m["日期"]]);
} catch (e) { /* 配置缺失就不显示倒计时 */ }
dv.table(["里程碑","日期","剩余"], milestones.map(([k, v]) => [k, v, Math.round(dv.date(v).diff(now, "days").days) + " 天"]));
dv.paragraph(`🔥 连续打卡 **${n}** 天${todayDone || n === 0 ? "" : "（截至昨天）"}　·　今天 **${doneToday}/${todayArr.length}** 项`);
```

**本周笔记**（周决策 + 打卡 + 本周的学习记录）

```dataview
TABLE WITHOUT ID
  file.link AS 笔记,
  choice(type = "session", "学习记录", choice(type = "weekly", "周决策", "打卡")) AS 类型,
  choice(type = "session",
         dateformat(date, "MM/dd"),
         dateformat(date(substring(range, 0, 10)), "MM/dd") + " ~ " + dateformat(date(substring(range, 13, 23)), "MM/dd")) AS 日期,
  choice(type = "session", topic, week) AS "主题 / 周"
FROM "05-日程安排/06-日志" OR "03-学习主题"
WHERE type = "weekly" OR type = "checkin"
   OR (type = "session" AND dateformat(date, "kkkk-'W'WW") = dateformat(date(today), "kkkk-'W'WW"))
SORT choice(type = "session", date, date(substring(range, 0, 10))) DESC
LIMIT 8
```

## 📊 各主题平均掌握度

```dataview

      choice(length(nonnull(rows.score)) > 0,
             round(average(nonnull(rows.score)), 1), "—") AS 平均掌握度
```

> [!note] 怎么让 AI 记住我的进度？
> AI 不会自动记住你 —— **你的笔记就是 AI 的记忆**。
> 每次对话前，打开当前主题的主页笔记并 @ 它，或用 `/更新进度 <主题名>`
> 让 AI 把最近的 Session 进度写回主页。AI 就能读到你的进度、水平和薄弱点。
